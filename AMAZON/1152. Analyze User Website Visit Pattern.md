# [1152. Analyze User Website Visit Pattern](https://leetcode.com/problems/analyze-user-website-visit-pattern/)

```java
class Solution {
    public class Visit {
        String username;
        int timestamp;
        String website;
    
        public Visit(String username, int timestamp, String website) {
            this.username = username;
            this.timestamp = timestamp;
            this.website = website;
        }
    }

    public List<String> mostVisitedPattern(String[] username, int[] timestamp, String[] website) {
        List<Visit> visits = new ArrayList<>();
        for (int i = 0; i < username.length; i++) {
            visits.add(new Visit(username[i], timestamp[i], website[i]));
        }

        // Sort asending on timestamp
        Collections.sort(visits, (a, b) -> a.timestamp - b.timestamp);

        // Each user's websites
        Map<String, List<String>> userWebsites = new HashMap<>();
        for (Visit v : visits) {
            userWebsites.computeIfAbsent(v.username, k -> new ArrayList<>()).add(v.website);
        }

        // Pattern counts
        int maxCount = 0;
        String result = "";
        Map<String, Integer> patternCounts = new HashMap<>();
        for (List<String> webs : userWebsites.values()) {
            if (webs.size() < 3) continue;
            Set<String> uniquePatterns = new HashSet<>();
            for (int i = 0; i < webs.size(); i++) {
                for (int j = i + 1; j < webs.size(); j++) {
                    for (int k = j + 1; k < webs.size(); k++) {
                        String p = webs.get(i) + "," + webs.get(j) + "," + webs.get(k);
                        if (uniquePatterns.add(p)) {
                            int count = patternCounts.getOrDefault(p, 0) + 1;
                            patternCounts.put(p, count);
                            if (count > maxCount || (count == maxCount && p.compareTo(result) < 0)) {
                                maxCount = count;
                                result = p;
                            }
                        }
                    }
                }
            }
        }
        return Arrays.asList(result.split(","));
    }
}
```
